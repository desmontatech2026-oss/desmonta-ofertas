function render() {

  $("feedTitle").textContent = conf.t;
  $("feedSub").textContent = conf.s;
  const all = listFor(state.tab).filter(matches);
  const page = all.slice(0, state.shown);
  $("feedCount").textContent = all.length + (state.tab === "cair" ? " no radar" : all.length === 1 ? " oferta" : " ofertas");
  if (!all.length) {
    const what = state.q ? `"${esc(state.q)}"` : esc(CATS[state.cat] || (TREND.find(t => t[0] === state.cat) || [0, "essa categoria"])[1]);

/* Tempo */
function updText() {
  if (!$("upd")) return;
  const mins = Math.max(0, Math.round((Date.now() - new Date(DATA.updated).getTime()) / 60000));
  let t;
  if (mins < 1) t = "atualizado agora";
  else if (mins < 60) t = `atualizado há ${mins} min`;
  else if (mins < 24 * 60) t = `atualizado há ${Math.round(mins / 60)} h`;
  else { const d = new Date(DATA.updated); t = `atualizado em ${String(d.getDate()).padStart(2, "0")}/${String(d.getMonth() + 1).padStart(2, "0")}`; }
  $("upd").textContent = t;
}
function tick() {
  const left = new Date(DATA.bf).getTime() - Date.now();
  if (left <= 0) {
    ["cdD","cdH","cdM","cdS"].forEach(id => { $(id).textContent = "00"; });
    $("bfNote").textContent = "Tá rolando! As ofertas mais quentes da Black Friday estão aqui embaixo.";
    return;
  }
  const s = Math.floor(left / 1000);
  $("cdD").textContent = Math.floor(s / 86400);
  $("cdH").textContent = String(Math.floor(s % 86400 / 3600)).padStart(2, "0");
  $("cdM").textContent = String(Math.floor(s % 3600 / 60)).padStart(2, "0");
  $("cdS").textContent = String(s % 60).padStart(2, "0");
}

function applyTheme(t) {
  if (t) document.documentElement.setAttribute("data-theme", t);

}
function currentTheme() {
  const a = document.documentElement.getAttribute("data-theme");
  if (a) return a;
  return window.matchMedia && matchMedia("(prefers-color-scheme: light)").matches ? "light" : "dark";
}

const SIZES = [7500, 9000, 12000, 18000, 22000, 24000, 30000, 36000];
function calcBTU() {
  const area = numDec($("bLarg").value) * numDec($("bComp").value);
  const pes = Math.max(1, Math.round(num($("bPes").value)));
  const ele = Math.max(0, Math.round(num($("bEle").value)));
  if (!area) { $("bRes").className = "res bad"; $("bRes").innerHTML = "<b>Coloca as medidas do cômodo</b>"; return; }
  const total = Math.round(area * ($("bSol").checked ? 800 : 600) + (pes - 1) * 600 + ele * 600);
  const size = SIZES.find(s => s >= total);
  $("bRes").className = "res";
  $("bRes").innerHTML = size
    ? `<b>Aparelho de ${size.toLocaleString("pt-BR")} BTU</b><p>Seu cômodo de ${area.toLocaleString("pt-BR", {maximumFractionDigits: 1})} m² pede uns ${total.toLocaleString("pt-BR")} BTU. Menos que isso, o aparelho trabalha no limite e gasta mais luz.</p><div class="row"><button class="btn sm" type="button" data-find="ar condicionado">Ver ar-condicionado em oferta</button></div>`
    : `<b>Mais de 36.000 BTU</b><p>Seu cômodo pede uns ${total.toLocaleString("pt-BR")} BTU. Nesse tamanho, compensa dividir em dois aparelhos.</p>`;
}
function calcPB() {
  const cap = num($("pCap").value), cel = num($("pCel").value);
  if (!cap || !cel) { $("pRes").className = "res bad"; $("pRes").innerHTML = "<b>Preenche as duas capacidades</b>"; return; }
  const c = Math.floor(cap * 0.65 / cel * 10) / 10;
  $("pRes").className = "res";
  $("pRes").innerHTML = `<b>Umas ${c.toLocaleString("pt-BR")} cargas completas</b><p>Um power bank de ${cap.toLocaleString("pt-BR")} mAh entrega na prática uns ${Math.round(cap * 0.65).toLocaleString("pt-BR")} mAh.</p>`;
}
function calcUN() {
  const u = $("uU").value;
  const base = u === "g" || u === "ml" ? 100 : 1;
  const name = {g:"100 g", ml:"100 ml", un:"unidade", m:"metro"}[u];
  const a = num($("uP1").value) / num($("uQ1").value) * base, b = num($("uP2").value) / num($("uQ2").value) * base;
  if (!isFinite(a) || !isFinite(b) || !a || !b) { $("uRes").className = "res bad"; $("uRes").innerHTML = "<b>Preenche preço e quantidade das duas</b>"; return; }
  const win = a <= b ? "A" : "B", diff = Math.round(Math.abs(a - b) / Math.max(a, b) * 100);
  $("uRes").className = "res";
  $("uRes").innerHTML = `<b>A oferta ${win} sai ${diff}% mais barata</b><p>A: ${brl(a)} por ${name}. B: ${brl(b)} por ${name}.</p>`;
}
function calcPC() {
  const g = GPUS[+$("cGpu").value];
  let w = num($("cW").value);
  const real = $("cGen").checked ? Math.round(w * 0.7) : w;
  if (!w) { $("cRes").className = "res bad"; $("cRes").innerHTML = "<b>Coloca a potência da sua fonte</b>"; return; }
  if (real >= g[1]) {
    $("cRes").className = "res";
    $("cRes").innerHTML = `<b>Aguenta a ${g[0]}</b><p>Fabricante pede ${g[1]} W. ${g[2]}: confere se sua fonte tem esse cabo.</p>`;
  } else {
    $("cRes").className = "res bad";
    $("cRes").innerHTML = `<b>Não aguenta a ${g[0]}</b><p>Fabricante pede ${g[1]} W e sua fonte entrega uns ${real} W. Pega uma fonte de ${Math.ceil(g[1] / 50) * 50} W ou mais com selo 80 Plus.</p><div class="row"><button class="btn sm" type="button" data-find="fonte">Ver fontes em oferta</button></div>`;
  }
}
function calcCabe() {
  const d = [num($("mL").value), num($("mA").value), num($("mP").value)].sort((x, y) => x - y);
  const dl = num($("dL").value), da = num($("dA").value), wp = num($("wP").value), ml = num($("mL").value);
  const msgs = [];
  let ok = true;
  if ($("mMont").checked) {
    if (d[0] <= dl - 2 && d[1] <= da - 2) msgs.push(`Passa na porta: a menor medida (${d[0]} cm) cabe no vão de ${dl} cm.`);
    else { ok = false; msgs.push(`Não passa na porta. A menor medida do móvel é ${d[0]} cm e o vão livre é ${dl} cm. Precisaria de pelo menos ${d[0] + 2} cm.`); }
  } else msgs.push("Vem desmontado: passa caixa por caixa. Só confere o espaço no cômodo.");
  if (wp) {
    if (ml <= wp) msgs.push(`Cabe na parede: sobram ${wp - ml} cm.`);
    else { ok = false; msgs.push(`Não cabe na parede: faltam ${ml - wp} cm.`); }
  }
  $("mRes").className = ok ? "res" : "res bad";
  $("mRes").innerHTML = `<b>${ok ? "Cabe" : "Não cabe"}</b>` + msgs.map(m => `<p>${m}</p>`).join("");
}

function copyFrom(el, ok) {
  const fallback = () => { const r = document.createRange(); r.selectNodeContents(el); const s = getSelection(); s.removeAllRanges(); s.addRange(r); toast("Texto selecionado. Copia pelo menu."); };

}

function calcVista() {
  const pix = num($("vPix").value), tot = num($("vParc").value), n = Math.max(1, Math.round(num($("vN").value))), r = num($("vR").value) / 100;
  if (!pix || !tot) { $("vRes").className = "res bad"; $("vRes").innerHTML = "<b>Preenche os dois preços</b>"; return; }
  const parc = tot / n, pv = r > 0 ? parc * (1 - Math.pow(1 + r, -n)) / r : tot;
  $("vRes").className = "res";
  $("vRes").innerHTML = pv < pix
    ? `<b>Parcelar sai melhor</b><p>Em ${n}x de ${brl(parc)}, com o dinheiro rendendo, o parcelado custa hoje o equivalente a ${brl(pv)}: ${brl(pix - pv)} a menos que o Pix.</p>`
    : `<b>Pagar no Pix sai melhor</b><p>O desconto do Pix vale mais que o rendimento. Você economiza uns ${brl(pv - pix)} pagando à vista.</p>`;
}
const TV_SIZES = [32, 40, 43, 50, 55, 58, 60, 65, 70, 75, 85, 98];
function calcTV() {
  const d = numDec($("tvD").value) * 100;
  if (!d) { $("tvRes").className = "res bad"; $("tvRes").innerHTML = "<b>Coloca a distância em metros</b>"; return; }
  const inch = d / 2.54;
  const fits = TV_SIZES.filter(x => x >= inch / 1.5 && x <= inch / 1.0);
  const ideal = TV_SIZES.reduce((a, b) => Math.abs(b - inch / 1.25) < Math.abs(a - inch / 1.25) ? b : a);

function explode(el) {
  if (!el) return;
  el.classList.remove("shine");
  const txt = el.textContent;
  el.innerHTML = [...txt].map(ch => ch === " " ? " " : `<span class="boom" style="--bx:${(Math.random() * 16 - 8).toFixed(1)}px;--by:${(Math.random() * 14 - 7).toFixed(1)}px;--br:${Math.round(Math.random() * 40 - 20)}deg">${esc(ch)}</span>`).join("");
  setTimeout(() => { el.textContent = txt; el.classList.add("shine"); }, 1000);
}
async function refreshData() {
  try {
    const r = await fetch("ofertas.json", {cache: "no-store"});
    if (!r.ok) return;
    const fresh = await r.json();
    if (fresh && fresh.updated && fresh.updated !== DATA.updated) {
      Object.assign(DATA, fresh);
      buildChips(); render(); renderCupons(); renderRel(); renderPed(); updText();
      toast("Chegaram ofertas novas");
    }
  } catch (e) {}
}

  $("nextT").textContent = Math.floor(nextIn / 60) + ":" + String(nextIn % 60).padStart(2, "0");
}
